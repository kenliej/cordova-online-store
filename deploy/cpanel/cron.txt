# cPanel cron jobs (Cron Jobs UI or `crontab -e`)
#
# Replace CPANEL_USER and PHP_BIN with your values.
# PHP_BIN is usually /usr/local/bin/php; use `which php` in cPanel Terminal.

# 1. Laravel scheduler — REQUIRED
#    Runs the hourly `orders:mark-unclaimed` command (claim-deadline expiry).
* * * * * PHP_BIN /home/CPANEL_USER/omc-backend/artisan schedule:run >> /home/CPANEL_USER/omc-backend/storage/logs/scheduler.log 2>&1

# 2. Queue worker (optional) — needed only for queued jobs/mails.
#    Password-reset emails are sent synchronously by default, so this can be
#    omitted. Enable if you set QUEUE_CONNECTION=database and queue notifications.
# */5 * * * * PHP_BIN /home/CPANEL_USER/omc-backend/artisan queue:work --stop-when-empty --tries=3 --max-time=280 >> /home/CPANEL_USER/omc-backend/storage/logs/queue.log 2>&1
